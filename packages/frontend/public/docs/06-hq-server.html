1
System Preparation
Update OS, create directories, configure firewall

1.1 Update System Packages

# Ubuntu / Debian
sudo apt update && sudo apt upgrade -y

# Install essential packages
sudo apt install -y curl wget git unzip ufw htop net-tools

1.2 Create Project Directory

# Clone the repository
cd /opt
sudo git clone https://github.com/your-org/TollGate-RFID.git
cd TollGate-RFID

1.3 Configure Firewall

# Enable UFW
sudo ufw enable

# Allow required ports
sudo ufw allow 22/tcp    # SSH
sudo ufw allow 80/tcp    # HTTP (Admin Panel)
sudo ufw allow 443/tcp   # HTTPS
sudo ufw allow 8080/tcp  # Customer Portal
sudo ufw allow 8081/tcp  # Financial Portal

# Deny direct DB access from external
sudo ufw deny 5432/tcp
sudo ufw deny 5433/tcp
sudo ufw deny 5434/tcp

# Reload
sudo ufw status
⚠️ Security: Never expose PostgreSQL ports (5432, 5433, 5434) to the public internet. Only the API port (3000) and frontend ports (80, 8080, 8081) should be accessible.
2
Docker Installation
Install Docker Engine and Docker Compose

2.1 Install Docker Engine

# Install Docker official GPG key
sudo install -m 0755 -d /etc/apt/keyrings
sudo curl -fsSL https://download.docker.com/linux/ubuntu/gpg -o /etc/apt/keyrings/docker.asc
sudo chmod a+r /etc/apt/keyrings/docker.asc

# Add Docker repository
echo \
  "deb [arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.asc] https://download.docker.com/linux/ubuntu \
  $(. /etc/os-release && echo "$VERSION_CODENAME") stable" | \
  sudo tee /etc/apt/sources.list.d/docker.list > /dev/null

# Install Docker
sudo apt update
sudo apt install -y docker-ce docker-ce-cli containerd.io docker-buildx-plugin docker-compose-plugin

# Verify installation
docker --version
docker compose version

2.2 Add User to Docker Group

# Allow running docker without sudo
sudo usermod -aG docker $USER

# Apply group changes (or log out and back in)
newgrp docker

# Verify
docker ps

    server_name admin.yourdomain.com;

Monitoring Script: Save the following as /opt/TollGate-RFID/scripts/monitor.sh and run via cron every 5 minutes for automated health checks. 
#!/bin/bash
# TollGate HQ Server Health Monitor
# Run: */5 * * * * /opt/TollGate-RFID/scripts/monitor.sh >> /var/log/tollgate-health.log 2>&1

LOG="/var/log/tollgate-health.log"
DATE=$(date '+%Y-%m-%d %H:%M:%S')

# Check API health
HEALTH=$(curl -s -o /dev/null -w "%{http_code}" http://localhost:3000/api/health/live)
if [ "$HEALTH" != "200" ]; then
    echo "[$DATE] CRITICAL: API health check failed (HTTP $HEALTH)" >> "$LOG"
    # Restart backend container
    docker restart tollgate-rfid-backend-1
fi

# Check each database
for DB_PORT in 5432 5433 5434; do
    if ! pg_isready -h localhost -p $DB_PORT -U postgres -q 2>/dev/null; then
        echo "[$date] CRITICAL: Database on port $DB_PORT unreachable" >> "$LOG"
    fi
done

# Check disk usage
DISK_USAGE=$(df / | awk 'NR==2 {print $5}' | tr -d '%')
if [ "$DISK_USAGE" -gt 85 ]; then
    echo "[$DATE] WARNING: Disk usage at ${DISK_USAGE}%" >> "$LOG"
fi

💾 Backup & Recovery

Automated backup scripts for all 3 PostgreSQL databases

B
Automated Backup Script
Save as /opt/TollGate-RFID/scripts/backup.sh — runs daily via cron
#!/bin/bash
# ============================================
# TollGate HQ Database Backup Script
# Backs up all 3 PostgreSQL databases
# ============================================

set -euo pipefail

# Configuration
BACKUP_DIR="/opt/TollGate-RFID/backups"
DATE=$(date '+%Y-%m-%d_%H-%M-%S')
RETENTION_DAYS=30

# Database configs
declare -A DATABASES=(
    ["hq"]="container=tollgate-rfid-db-1 db=tollgate"
    ["customer"]="container=tollgate-rfid-customer-db-1 db=tollgate_customer"
    ["plaza"]="container=tollgate-rfid-plaza-db-1 db=tollgate_plaza"
)

# Create backup directory
mkdir -p "$BACKUP_DIR"

# Backup each database
for DB_NAME in "${!DATABASES[@]}"; do
    IFS='=' read -r CONTAINER_VAR CONTAINER DB_VAR DB <<< "${DATABASES[$DB_NAME]}"
    CONTAINER=$(echo "$CONTAINER" | tr -d ' "')
    DB=$(echo "$DB" | tr -d ' "')

    BACKUP_FILE="$BACKUP_DIR/${DB_NAME}_${DATE}.sql.gz"

    echo "[$(date)] Backing up $DB from $CONTAINER..."
    docker exec "$CONTAINER" pg_dump -U postgres "$DB" | gzip > "$BACKUP_FILE"

    echo "[$(date)] Backup saved: $BACKUP_FILE ($(du -h "$BACKUP_FILE" | cut -f1))"
done

# Cleanup old backups (older than RETENTION_DAYS)
echo "[$(date)] Cleaning backups older than $RETENTION_DAYS days..."
find "$BACKUP_DIR" -name "*.sql.gz" -mtime +$RETENTION_DAYS -delete

# Create backup manifest
echo "[$(date)] Backup manifest:" > "$BACKUP_DIR/manifest.txt"
ls -lh "$BACKUP_DIR"/*.sql.gz | tail -10 >> "$BACKUP_DIR/manifest.txt"

echo "[$(date)] All backups completed successfully."

Setup Cron Job

# Make executable
chmod +x /opt/TollGate-RFID/scripts/backup.sh

# Run daily at 2:00 AM
sudo crontab -e
# Add: 0 2 * * * /opt/TollGate-RFID/scripts/backup.sh >> /var/log/tollgate-backup.log 2>&1

💡 Troubleshooting

Common issues and their solutions

IssueSymptomsDiagnosisFix
DB Connection Refused Backend logs show "connection refused" docker ps — check DB container status docker compose up -d db customer-db plaza-db
Backend 502 Bad Gateway Frontend shows 502 on API calls docker logs tollgate-rfid-backend-1 --tail 50 Check DB health, then restart: docker restart backend
Migration Not Applied Missing tables, Prisma errors docker exec backend npx prisma migrate status docker exec backend npx prisma migrate deploy
Prisma Client Outdated "Prisma Client is not generated" error docker exec backend npx prisma generate Regenerate: npx prisma generate in backend container
Rate Limit 429 Error "Too Many Requests" response In-memory rate limiter hit threshold Wait for window reset or restart backend: docker restart backend
CORS Error Browser console shows CORS policy block Check CORS_ORIGINS env var Update .env with correct origins, restart backend
Frontend 404 on Reload SPA routes return 404 after page refresh Nginx SPA fallback not configured Ensure try_files $uri $uri/ /index.html in nginx.conf
Disk Space Full Containers crash, "no space left on device" docker system df — check volumes docker system prune -f + clean old backups
Customer Portal Login Failed "Login failed" on valid credentials API returns {user, token} directly Use res.data not res.data.data in frontend
Sync Queue Stuck Plaza events not reaching HQ GET /api/sync/status on plaza Check network connectivity, restart plaza sync service
